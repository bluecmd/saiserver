# crate_universe resolves metadata using downloaded host tools, not PATH cargo
# or rustc. Use a unique repository name: rules_rust itself also declares a
# rust_host_tools repository. This extension is distinct from action toolchains.
rust_host = use_extension("@rules_rust//rust:extensions.bzl", "rust_host_tools")
rust_host.host_tools(
    name = "saiserver_rust_host_tools",
    version = "1.94.0",
    rustfmt_version = "1.94.0",
    sha256s = SAISERVER_RUST_SHA256S,
)
use_repo(rust_host, "saiserver_rust_host_tools")

crate = use_extension("@rules_rust//crate_universe:extensions.bzl", "crate")
crate.from_cargo(
    name = "saiserver_crates",
    cargo_lockfile = "//tui:Cargo.lock",
    manifests = ["//tui:Cargo.toml"],
    host_tools = "@saiserver_rust_host_tools//:BUILD.bazel",
    isolated = True,
    skip_cargo_lockfile_overwrite = True,
    supported_platform_triples = ["x86_64-unknown-linux-gnu"],
)
use_repo(crate, "saiserver_crates")

# The parent must also provide a root BUILD.bazel (even an empty file):
# crate_universe 0.75.0 resolves the root //:MODULE.bazel label internally.
# Parent IDL contract: //src:sai_tui_rs must be visible to //tui and expose
# exactly one generated .rs file from the declared, downloaded Thrift compiler.
# No generator invocation is embedded in these Rust rules.
#
# Public API (also usable as @saiserver//tui:... by downstream modules):
#   //tui:sai_tui       Rust library, crate name sai_tui (checked-in RPC module)
#   //tui:sai-tui       executable
#   //tui:sai_tui_rpc   standalone generated library, crate name sai_tui_rpc
#   //tui:tests         library unit tests plus cli_test and safety_test
#
# Integration verification after the parent installs MODULE/native rules:
#   bazel build //tui:sai_tui //tui:sai-tui //tui:sai_tui_rpc
#   bazel test //tui:tests --test_output=errors
# Also test from a downstream local_path_override consumer to verify external
# repository runfiles paths. cli_test declares :sai-tui in data and embeds its
# $(rootpath), which remains relative to the main repository's test runfiles
# working directory (including ../<external-repository>/... when consumed).
